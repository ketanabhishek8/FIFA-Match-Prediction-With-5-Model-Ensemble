# Runtime dependencies: what the API needs to serve predictions, and all that
# the Vercel deployment installs (see docs/DEPLOYMENT.md). Development tools --
# data collection, figures, notebooks, tests, the local server -- are in
# requirements-dev.txt.
#
# Pinned exactly, not with >=: the served model bundle is a pickle, and loading
# it under a different scikit-learn / xgboost / numpy / pandas than the one
# that wrote it can fail outright or quietly change predictions. Bump these
# together with a retrained bundle.
fastapi==0.141.1
numpy==2.5.3
pandas==3.0.5
pyyaml==6.0.3
scikit-learn==1.9.1
# Imported directly (evaluate.py, score_model.py, train.py, model_cache.py),
# not just as scikit-learn transitives.
scipy==1.18.1
joblib==1.6.0
# On Linux the standard xgboost wheel pulls in ~250 MB of NVIDIA GPU libraries,
# enough on its own to push the Vercel bundle past its 500 MB limit. xgboost-cpu
# is the same library under the same `import xgboost`, without them. It is only
# published for Linux and Windows, so other platforms keep the standard package.
xgboost-cpu==3.4.1; sys_platform == "linux"
xgboost==3.4.1; sys_platform != "linux"
